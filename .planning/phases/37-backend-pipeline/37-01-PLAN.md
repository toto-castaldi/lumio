import { serve } from "https://deno.land/std@0.177.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers":
    "authorization, x-client-info, apikey, content-type",
};

function createUserSupabaseClient(authHeader: string) {
  const supabaseUrl = Deno.env.get("SUPABASE_URL")!;
  const supabaseAnonKey = Deno.env.get("SUPABASE_ANON_KEY")!;
  return createClient(supabaseUrl, supabaseAnonKey, {
    global: { headers: { Authorization: authHeader } },
  });
}

async function getUserId(supabase: ReturnType<typeof createClient>): Promise<string> {
  const { data: { user }, error } = await supabase.auth.getUser();
  if (error || !user) throw new Error("Unauthorized");
  return user.id;
}

    `validateUserPath(userId: string, filePath: string): string`:

    Environment variables (from Deno.env.get):
    - GITHUB_PAT -- personal access token with repo/Contents scope
    - GITHUB_REPO_OWNER -- GitHub account owning the shared repo
    - GITHUB_REPO_NAME -- shared repo name

    Helper functions (copy pattern from git-sync):
    - `createUserSupabaseClient(authHeader)` -- creates Supabase client with user JWT
    - `getUserId(supabase)` -- extracts user.id from auth
    - `validateUserPath(userId, filePath)` -- path isolation (same logic as tests)
    - `encodeContent(content)` -- UTF-8 safe base64

    GitHub API helpers:
    - `githubFetch(path, options?)` -- wraps fetch to `https://api.github.com/repos/${OWNER}/${REPO}/contents/${path}` with auth headers (`Bearer ${GITHUB_PAT}`, Accept: `application/vnd.github+json`, X-GitHub-Api-Version: `2022-11-28`)
    - `getFile(path)` -- GET file, returns `{ content, sha, name, path }` or null for 404
    - `commitFile(path, content, message, sha?)` -- PUT file (base64 encoded), sha required for updates
    - `deleteFile(path, sha, message)` -- DELETE file with sha
    - `listDirectory(path)` -- GET directory, returns array of `{ name, path, sha, type, size }`

    Actions (action-based routing via JSON body):

    1. `commit_file`: Body `{ action, path, content, sha?, message? }`. Validates path against userId. Calls commitFile. Returns `{ success: true, sha, commit_sha }`.

    2. `delete_file`: Body `{ action, path, sha }`. Validates path. Calls deleteFile. Returns `{ success: true }`.

    3. `get_file`: Body `{ action, path }`. Validates path. Calls getFile. Returns `{ success: true, content, sha, name, path }` or 404.

    4. `list_files`: Body `{ action, path }`. Validates path (must be `{userId}/` or `{userId}/{deck}/`). Calls listDirectory. Returns `{ success: true, files: [...] }`.

    5. `list_decks`: Body `{ action }`. Lists `{userId}/` directory (no path validation needed -- always scoped to user). Returns `{ success: true, decks: [{ name, path }] }` filtering only `type: "dir"`.

    Error handling:
    - Missing Authorization header -> 401
    - Invalid action -> 400
    - Path validation failure -> 403
    - GitHub API errors -> proxy status code and message
    - Catch-all -> 500 with error message

    CORS: Handle OPTIONS preflight with corsHeaders (same as git-sync).

    Commit messages should be descriptive: `[deck-builder] Create {path}`, `[deck-builder] Update {path}`, `[deck-builder] Delete {path}`.

    ```typescript
    export async function createDeck(name: string): Promise<DeckEntry> {
      const data = await invoke<{ deck: DeckEntry }>({ action: 'create_deck', name });
      return data.deck;
    }

    export async function renameDeck(oldName: string, newName: string): Promise<DeckEntry> {
      const data = await invoke<{ deck: DeckEntry }>({ action: 'rename_deck', old_name: oldName, new_name: newName });
      return data.deck;
    }

    export async function deleteDeck(name: string): Promise<void> {
      await invoke<{ success: true }>({ action: 'delete_deck', name });
    }
    ```

    **Part B -- Edge function actions:**
    Add three new cases to the switch in supabase/functions/deck-commit/index.ts, BEFORE the `default` case:

    **create_deck:**
    - Validate `name` field exists and is a string
    - Validate deck name format: `/^[a-zA-Z0-9][a-zA-Z0-9 -]*$/`, trimmed, max 50 chars, not reserved
    - Construct path: `${userId}/${name.trim()}`
    - Check if deck already exists via `listDirectory(deckPath)` -- if items returned, return 409 "Deck already exists"
    - Commit `.gitkeep` file at `${deckPath}/.gitkeep` with empty content and message `[deck-builder] Create deck: ${name.trim()}`
    - Use `commitFile()` directly (bypasses validateUserPath since .gitkeep is not .md -- this is safe because path is constructed server-side from validated userId + validated name)
    - Return `{ success: true, deck: { name: name.trim(), path: deckPath } }`

    **rename_deck:**
    - Validate `old_name` and `new_name` fields
    - Validate new_name format (same regex as create)
    - Construct old and new paths: `${userId}/${old_name.trim()}`, `${userId}/${new_name.trim()}`
    - Check old deck exists via `listDirectory(oldPath)` -- if empty, return 404 "Deck not found"
    - Check new deck does NOT exist via `listDirectory(newPath)` -- if items, return 409 "Deck already exists"
    - For each file in old directory: get file content via `getFile(file.path)`, create at new path via `commitFile(newPath/file.name, content, message)`, delete old via `deleteFile(file.path, file.sha, message)`
    - Return `{ success: true, deck: { name: new_name.trim(), path: newPath } }`

    **delete_deck:**
    - Validate `name` field
    - Construct path: `${userId}/${name.trim()}`
    - List all files via `listDirectory(deckPath)`
    - If empty, return 404 "Deck not found"
    - Delete each file (need sha from listDirectory response -- it includes sha)
    - Return `{ success: true }`

    IMPORTANT: For the rename operation, files must be processed sequentially (not Promise.all) because GitHub Contents API can have conflicts with concurrent operations on the same repo.

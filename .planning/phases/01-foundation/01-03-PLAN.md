  <action>
Create `apps/android/components/ConnectionTest.tsx` that verifies Supabase connectivity.

IMPORTANT: Use `auth.getSession()` instead of querying `platform_config` table.
The `platform_config` table has RLS policy `TO authenticated` which requires the user to be logged in.
Since this is Phase 1 (pre-authentication), we need a test that works without auth.
`auth.getSession()` works pre-authentication and confirms:
1. Network connectivity to Supabase
2. Supabase client is configured correctly
3. Anon key is valid

```typescript
import { useEffect, useState } from 'react';
import { Text, View, ActivityIndicator } from 'react-native';
import { supabase } from '../lib/supabase';

type ConnectionStatus = 'loading' | 'connected' | 'error';

/**
 * ConnectionTest - Verifies Supabase backend connectivity
 *
 * Uses auth.getSession() to verify:
 * 1. Network connectivity to Supabase
 * 2. Supabase client is configured correctly
 * 3. Anon key is valid
 *
 * Note: Does NOT query tables because platform_config requires authentication.
 * auth.getSession() works pre-authentication and returns null session (expected).
 *
 * Used for Phase 1 success criteria verification.
 */
export function ConnectionTest() {
  const [status, setStatus] = useState<ConnectionStatus>('loading');
  const [error, setError] = useState<string | null>(null);
  const [latency, setLatency] = useState<number | null>(null);

  useEffect(() => {
    async function testConnection() {
      const startTime = Date.now();
      try {
        // Use auth.getSession() - works pre-authentication
        // Returns { data: { session: null }, error: null } when not logged in
        // This confirms Supabase client is configured and reachable
        const { error: sessionError } = await supabase.auth.getSession();

        if (sessionError) {
          throw sessionError;
        }

        setLatency(Date.now() - startTime);
        setStatus('connected');
      } catch (e) {
        setStatus('error');
        setError(e instanceof Error ? e.message : 'Unknown error');
      }
    }

    testConnection();
  }, []);

  return (
    <View className="mx-4 p-4 bg-white rounded-lg shadow-sm">
      <Text className="text-lg font-semibold text-gray-900">
        Supabase Connection
      </Text>

      <View className="flex-row items-center mt-2">
        {status === 'loading' && (
          <>
            <ActivityIndicator size="small" color="#6B7280" />
            <Text className="text-gray-500 ml-2">Connecting...</Text>
          </>
        )}

        {status === 'connected' && (
          <>
            <View className="w-3 h-3 rounded-full bg-green-500" />
            <Text className="text-green-600 ml-2 font-medium">Connected</Text>
            {latency !== null && (
              <Text className="text-gray-400 ml-2">({latency}ms)</Text>
            )}
          </>
        )}

        {status === 'error' && (
          <>
            <View className="w-3 h-3 rounded-full bg-red-500" />
            <Text className="text-red-600 ml-2 font-medium">Error</Text>
          </>
        )}
      </View>

      {error && (
        <View className="mt-2 p-2 bg-red-50 rounded">
          <Text className="text-red-700 text-sm">{error}</Text>
        </View>
      )}

      <Text className="text-gray-400 text-xs mt-3">
        Testing connection to Supabase backend
      </Text>
    </View>
  );
}
```

AVOID: Querying platform_config table (requires authentication - RLS policy "TO authenticated").
AVOID: Blocking the UI while connection test runs (use useEffect).

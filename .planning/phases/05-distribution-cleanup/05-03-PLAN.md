**Add `build-apk` job** after `lint-and-typecheck`, before `deploy-migrations`:

```yaml
build-apk:
  runs-on: ubuntu-latest
  needs: [lint-and-typecheck, auto-release]
  if: always() && github.ref == 'refs/heads/main' && github.event_name == 'push' && needs.lint-and-typecheck.result == 'success' && needs.auto-release.outputs.released == 'true'
  steps:
    - uses: actions/checkout@v4
      with:
        ref: main
        fetch-depth: 0

    - name: Pull latest changes
      run: git pull origin main

    - uses: actions/setup-java@v4
      with:
        distribution: 'temurin'
        java-version: '17'

    - uses: pnpm/action-setup@v4

    - uses: actions/setup-node@v4
      with:
        node-version: '22'
        cache: 'pnpm'

    - name: Cache Gradle
      uses: actions/cache@v4
      with:
        path: |
          ~/.gradle/caches
          ~/.gradle/wrapper
        key: gradle-${{ hashFiles('apps/android/android/gradle/wrapper/gradle-wrapper.properties') }}
        restore-keys: gradle-

    - name: Install dependencies
      run: pnpm install --frozen-lockfile

    - name: Build packages
      run: pnpm build:packages

    - name: Expo prebuild
      working-directory: apps/android
      run: npx expo prebuild --platform android --clean
      env:
        EXPO_PUBLIC_SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
        EXPO_PUBLIC_SUPABASE_ANON_KEY: ${{ secrets.SUPABASE_ANON_KEY }}
        EXPO_PUBLIC_GOOGLE_WEB_CLIENT_ID: ${{ secrets.GOOGLE_WEB_CLIENT_ID }}

    - name: Decode keystore
      run: echo "${{ secrets.ANDROID_KEYSTORE_BASE64 }}" | base64 --decode > apps/android/android/app/release.keystore

    - name: Build release APK
      working-directory: apps/android/android
      run: ./gradlew app:assembleRelease -PversionCode=${{ github.run_number }} -PversionName=${{ needs.auto-release.outputs.new_version }}
      env:
        ANDROID_KEYSTORE_PASSWORD: ${{ secrets.ANDROID_KEYSTORE_PASSWORD }}
        ANDROID_KEY_ALIAS: ${{ secrets.ANDROID_KEY_ALIAS }}
        ANDROID_KEY_PASSWORD: ${{ secrets.ANDROID_KEY_PASSWORD }}

    - name: Rename APK
      run: cp apps/android/android/app/build/outputs/apk/release/app-release.apk lumio.apk

    - name: Upload to GitHub Release
      uses: softprops/action-gh-release@v2
      with:
        tag_name: v${{ needs.auto-release.outputs.new_version }}
        files: lumio.apk
```

Key points:
- Job runs only on push to main AND when auto-release created a new version (`needs.auto-release.outputs.released == 'true'`)
- EXPO_PUBLIC env vars set during prebuild step so they get baked into the native build
- Keystore decoded from base64 secret before Gradle build
- Version code from `github.run_number`, version name from auto-release output
- APK renamed to `lumio.apk` for consistent download URL
- Uploaded to the GitHub Release tag created by auto-release

**IMPORTANT:** The `if` condition uses `always() &&` to run even when auto-release is skipped (e.g., on PRs). The `needs.auto-release.outputs.released == 'true'` check ensures APK only builds when there's a new release. This is the same pattern used by existing deploy jobs.

Add `deploy-landing` job to `.github/workflows/ci-deploy.yml`, in the deploy section (after build-apk, near where deploy-web was):

```yaml
deploy-landing:
  runs-on: ubuntu-latest
  needs: [lint-and-typecheck, auto-release]
  if: always() && github.ref == 'refs/heads/main' && github.event_name == 'push' && needs.lint-and-typecheck.result == 'success'
  steps:
    - uses: actions/checkout@v4
      with:
        ref: main

    - name: Deploy to DigitalOcean
      uses: appleboy/scp-action@v0.1.7
      with:
        host: ${{ secrets.DO_HOST }}
        username: ${{ secrets.DO_USERNAME }}
        key: ${{ secrets.DO_SSH_KEY }}
        source: 'apps/landing/*'
        target: '/var/www/lumio'
        strip_components: 2

    - name: Reload Nginx
      uses: appleboy/ssh-action@v1.0.3
      with:
        host: ${{ secrets.DO_HOST }}
        username: ${{ secrets.DO_USERNAME }}
        key: ${{ secrets.DO_SSH_KEY }}
        script: sudo systemctl reload nginx
```

Key differences from old deploy-web:
- No build step needed (landing page is static HTML, no Vite build)
- `source: 'apps/landing/*'` instead of `apps/web/dist/*`
- `strip_components: 2` (strips `apps/landing/` prefix, deploying contents to `/var/www/lumio/`)
- Runs on every push to main (not gated on auto-release, since the landing page should always be up to date)
- No pnpm/node setup needed (no build step)

      name: string;
      path: string;
    }

    Create a private helper:
    ```typescript
    async function invoke<T>(body: Record<string, unknown>): Promise<T> {
      const { data, error } = await supabase.functions.invoke('deck-commit', { body });
      if (error) throw new Error(error.message || 'Edge function error');
      if (data?.error) throw new Error(data.error);
      return data as T;
    }
    ```

    Export 5 functions:
    - `commitFile(path: string, content: string, sha?: string, message?: string): Promise<CommitResult>` -- calls invoke with `{ action: "commit_file", path, content, sha, message }`

    - `getFile(path: string): Promise<FileContent>` -- calls invoke with `{ action: "get_file", path }`
    - `listFiles(path: string): Promise<FileEntry[]>` -- calls invoke with `{ action: "list_files", path }`, returns `data.files`
    - `listDecks(): Promise<DeckEntry[]>` -- calls invoke with `{ action: "list_decks" }`, returns `data.decks`

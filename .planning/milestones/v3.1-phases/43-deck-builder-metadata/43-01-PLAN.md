    2. Trims and validates deck name via `validateDeckName()`
    3. Constructs path as `${userId}/${trimmedDeckName}/deck.yaml` (bypasses validateUserPath, same pattern as commit_yaml)
    4. Calls internal `getFile(yamlPath)`
    5. Returns 404 with `{ error: "File not found" }` if null
    6. Returns 200 with `{ success: true, content: file.content, sha: file.sha }` if found

    Follow the exact response format pattern used by other cases (corsHeaders, Content-Type header, JSON.stringify).

    **Client (api.ts):** Add two exported functions and one exported interface:

    ```typescript
    export interface DeckMetadata {
      display_name: string;
      description: string;
      tags: string[];
      language: string;
    }

    export async function getDeckYaml(deckName: string): Promise<{ content: string; sha: string } | null> {
      try {
        const data = await invoke<{ content: string; sha: string }>({
          action: 'get_yaml', deck_name: deckName,
        });
        return { content: data.content, sha: data.sha };
      } catch (err) {
        if (err instanceof Error && err.message === 'File not found') return null;
        throw err;
      }
    }

    export async function commitYaml(deckName: string, metadata: DeckMetadata): Promise<CommitResult> {
      const data = await invoke<{ sha: string; commit_sha: string }>({
        action: 'commit_yaml',
        deck_name: deckName,
        ...metadata,
      });
      return { sha: data.sha, commit_sha: data.commit_sha };
    }
    ```

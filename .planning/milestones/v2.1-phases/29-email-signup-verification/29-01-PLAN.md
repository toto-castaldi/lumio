Implementation:
```typescript
const [verifyLoading, setVerifyLoading] = useState(false);
const [resendLoading, setResendLoading] = useState(false);

const verifyEmailOtp = useCallback(async (email: string, token: string): Promise<void> => {
  setVerifyLoading(true);
  try {
    const { error } = await getSupabaseClient().auth.verifyOtp({
      email,
      token,
      type: 'email',  // NOT 'signup' (deprecated)
    });
    if (error) throw error;
    // onAuthStateChange fires SIGNED_IN, session auto-updates, state -> 'ready'
  } finally {
    setVerifyLoading(false);
  }
}, []);

const resendOtp = useCallback(async (email: string): Promise<void> => {
  setResendLoading(true);
  try {
    const { error } = await getSupabaseClient().auth.resend({
      type: 'signup',  // resend uses 'signup', NOT 'email'
      email,
    });
    if (error) throw error;
  } finally {
    setResendLoading(false);
  }
}, []);
```

Add both methods and loading states to the context value object. Add default values (async no-ops for methods, false for loading states) to the createContext default.
